#   pip install ".[rag]"        # 🧠 RAG queries + spaCy NER
#   pip install ".[full]"       # everything above
#

#
# NOTE (V0.13 audit): the former `kokoro>=0.9.0` / `"misaki[en]">=0.9.0`

#      https://github.com/oschwartz10612/poppler-windows/releases
